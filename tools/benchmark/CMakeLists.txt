cmake_minimum_required(VERSION 3.24)
project(path_tracer_benchmark LANGUAGES CXX CUDA)
include(FetchContent)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_GTEST_TESTS OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(googlebenchmark
    URL https://codeload.github.com/google/benchmark/tar.gz/192ef10025eb2c4cdd392bc502f0c852196baa48
    URL_HASH SHA256=f82705a2726d8f6cdcda274b841f6314dbfc6f731cdda06c946f310ec1cc3ad9)
FetchContent_MakeAvailable(googlebenchmark)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../wavefront.cu" source_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../benchmark_main.cu" adapter_hash)
function(workload name width height samples scene)
    add_executable(${name} ../benchmark_main.cu)
    target_link_libraries(${name} PRIVATE benchmark::benchmark)
    target_compile_features(${name} PRIVATE cxx_std_17 cuda_std_17)
    target_compile_definitions(${name} PRIVATE PT_WIDTH=${width} PT_HEIGHT=${height} PT_SAMPLES=${samples} PT_SCENE=${scene} PT_SOURCE_HASH="${source_hash}" PT_ADAPTER_HASH="${adapter_hash}")
    target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-O2>)
    set_target_properties(${name} PROPERTIES CUDA_ARCHITECTURES 120)
endfunction()
